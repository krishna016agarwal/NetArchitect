! ===== DEVICE: RTR-HQ =====
!
hostname RTR-HQ
no ip domain-lookup
!
! --- VLAN Configuration (IEEE 802.1Q) ---
vlan 10
 name HQ-Users
vlan 20
 name Colo-Servers
vlan 999
 name NATIVE
!
! --- Spanning Tree (IEEE 802.1w RSTP) ---
spanning-tree mode rapid-pvst
spanning-tree vlan 10 priority 4096
spanning-tree vlan 20 priority 4096
!
! --- Layer 3 Routing ---
ip routing
!
! --- SVI Interfaces ---
interface Vlan10
 description HQ-Users SVI
 ip address 172.16.0.1 255.255.255.0
 no shutdown
!
interface Vlan20
 description Colo-Servers SVI
 ip address 172.16.1.1 255.255.255.128
 no shutdown
!
! --- Access Ports ---
interface GigabitEthernet1/0/11
 description HQ-Users access
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
interface GigabitEthernet1/0/21
 description Colo-Servers access
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! --- Routed Point-to-Point Ports ---
interface GigabitEthernet0/1
 description P2P to R1
 no switchport
 ip address 172.16.2.129 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/2
 description P2P to R2
 no switchport
 ip address 172.16.2.133 255.255.255.252
 no shutdown
!
! --- Loopback Interfaces ---
interface Loopback0
 description Router-ID RTR-HQ
 ip address 172.16.2.140 255.255.255.255
 no shutdown
!
! --- Dot1Q Subinterfaces (Metro EVPL handoff) ---
interface GigabitEthernet0/1.10
 description EVPL HQ-Users
 encapsulation dot1Q 10
 ip address 172.16.0.1 255.255.255.0
 no shutdown
!
interface GigabitEthernet0/1.20
 description EVPL Colo-Servers
 encapsulation dot1Q 20
 ip address 172.16.1.1 255.255.255.128
 no shutdown
!
! --- OSPF Configuration (RFC 2328) ---
router ospf 1
 router-id 172.16.2.140
 network 172.16.0.0 0.0.0.255 area 0
 network 172.16.1.0 0.0.0.127 area 0
 network 172.16.2.128 0.0.0.3 area 0
 network 172.16.2.132 0.0.0.3 area 0
 network 172.16.2.140 0.0.0.0 area 0
 passive-interface Vlan10
 passive-interface Vlan20
!
! --- QoS Configuration (Metro) ---
class-map match-all VOICE-CLASS
 match dscp ef
!
policy-map QOS-POLICY
 class VOICE-CLASS
  priority percent 30
!
interface GigabitEthernet0/1
 service-policy output QOS-POLICY
!
end
